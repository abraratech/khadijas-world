  235: export function findAnimationName(
  236:   availableNames: readonly string[],
  237:   candidates: readonly string[] | undefined,
  238: ): string | null {
  239:   if (!candidates?.length) return null;
  240:   const exact = new Map(availableNames.map((name) => [name.toLowerCase(), name]));
  241:   for (const candidate of candidates) {
  242:     const match = exact.get(candidate.toLowerCase());
  243:     if (match) return match;
  244:   }
  245:   for (const candidate of candidates) {
  246:     const loweredCandidate = candidate.toLowerCase();
  247:     const partial = availableNames.find((name) => name.toLowerCase().includes(loweredCandidate));
  248:     if (partial) return partial;
  249:   }
  250:   return null;
  251: }
